cmake_minimum_required(VERSION 3.8)
project(smartcar_posegraph_reader)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization)
find_package(Eigen3 REQUIRED)
get_filename_component(ROS_PREFIX "${rclcpp_DIR}/../../.." ABSOLUTE)
find_path(KARTO_INCLUDE_DIR karto_sdk/Mapper.h HINTS "${ROS_PREFIX}/include" REQUIRED)
find_library(KARTO_LIBRARY kartoSlamToolbox HINTS "${ROS_PREFIX}/lib" REQUIRED)
add_executable(inspect_posegraph inspect_posegraph.cpp)
target_compile_features(inspect_posegraph PUBLIC cxx_std_17)
target_include_directories(inspect_posegraph PRIVATE ${KARTO_INCLUDE_DIR})
target_link_libraries(inspect_posegraph ${KARTO_LIBRARY} Boost::serialization Eigen3::Eigen)
ament_target_dependencies(inspect_posegraph rclcpp)
