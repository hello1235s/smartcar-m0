cmake_minimum_required(VERSION 3.8)
project(smartcar_nav_checks)
find_package(ament_cmake REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_util REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
add_library(valid_pose_goal_checker SHARED src/valid_pose_goal_checker.cpp)
target_compile_features(valid_pose_goal_checker PUBLIC cxx_std_17)
target_include_directories(valid_pose_goal_checker PRIVATE include)
ament_target_dependencies(valid_pose_goal_checker nav2_core nav2_util pluginlib rclcpp rclcpp_lifecycle geometry_msgs nav2_costmap_2d)
add_library(ackermann_rpp SHARED src/ackermann_rpp.cpp)
target_compile_features(ackermann_rpp PUBLIC cxx_std_17)
target_include_directories(ackermann_rpp PRIVATE include)
ament_target_dependencies(ackermann_rpp nav2_core nav2_util pluginlib rclcpp rclcpp_lifecycle geometry_msgs nav2_costmap_2d nav2_regulated_pure_pursuit_controller)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS valid_pose_goal_checker ackermann_rpp LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(valid_pose_test test/valid_pose_test.cpp)
  target_compile_features(valid_pose_test PUBLIC cxx_std_17)
  target_include_directories(valid_pose_test PRIVATE include)
  add_test(NAME valid_pose_test COMMAND valid_pose_test)
  add_executable(ackermann_limits_test test/ackermann_limits_test.cpp)
  target_compile_features(ackermann_limits_test PUBLIC cxx_std_17)
  target_include_directories(ackermann_limits_test PRIVATE include)
  add_test(NAME ackermann_limits_test COMMAND ackermann_limits_test)
  add_executable(plugin_loading_test test/plugin_loading_test.cpp)
  target_compile_features(plugin_loading_test PUBLIC cxx_std_17)
  ament_target_dependencies(plugin_loading_test nav2_core pluginlib rclcpp)
  add_test(NAME plugin_loading_test COMMAND plugin_loading_test)
  add_executable(tf2_lock_order_test test/tf2_lock_order_test.cpp)
  target_compile_features(tf2_lock_order_test PUBLIC cxx_std_17)
  ament_target_dependencies(tf2_lock_order_test tf2 geometry_msgs)
  add_test(NAME tf2_lock_order_test COMMAND tf2_lock_order_test)
  get_filename_component(task_root "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
  set_tests_properties(tf2_lock_order_test PROPERTIES TIMEOUT 5)
  set_tests_properties(tf2_lock_order_test plugin_loading_test PROPERTIES
    ENVIRONMENT "LD_LIBRARY_PATH=${task_root}/vendor/m3-dds/opt/ros/humble/lib:$ENV{LD_LIBRARY_PATH}")
endif()
ament_package()
