cmake_minimum_required(VERSION 3.8)
project(dashboard_tools)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
add_executable(dashboard_aggregator src/aggregator.cpp)
ament_target_dependencies(dashboard_aggregator rclcpp std_msgs rcl_interfaces)
install(TARGETS dashboard_aggregator DESTINATION lib/${PROJECT_NAME})
ament_package()
